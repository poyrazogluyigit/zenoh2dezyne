cpg.call.name("[declare_subscriber]|[declare_publisher]") -> CALL nodes for each pub/sub
-> zenoh libs must be in project root dir for including zenoh in the cpg

// List all non-library include source files
cpg.file.name.filterNot(_.contains("joern_includes")).label

// Get Lambda CFGs
cpg.call.name("declare_subscriber").map { subCall =>
  val topic = subCall.argument(1).code
  val cbArgCode = subCall.argument(2).code
  val cbVarName = cbArgCode.replace("&", "").trim

  val resolvedGraph =
    {
      val directCfg = cpg.method.name(cbVarName).dotCfg.l
      if (directCfg.nonEmpty) {
        directCfg
      } else {
        cpg.assignment
          .where(_.argument(1).codeExact(cbVarName))
          .argument(2)
          .ast.isMethodRef
          .referencedMethod
          .dotCfg
          .l
      }
    }

  Map(
    "topic" -> topic,
    "callback" -> cbVarName,
    "dotGraph" -> resolvedGraph.headOption.getOrElse("CFG resolution failed")
  )
}.toJsonPretty
